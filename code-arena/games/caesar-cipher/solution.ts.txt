// Le Message secret - solution de reference (TypeScript)
//
// On ne connait pas le decalage : on essaie les 26. Pour chacun, on dechiffre
// (chaque lettre recule de k rangs, modulo 26 pour boucler de A vers Z) et on
// garde celui qui fait apparaitre l'indice dans le message en clair.
//
// Notes sur le runtime TypeScript de Piston : il compile avec la bibliotheque ES5
// (pas de String.includes, d'ou indexOf) et sans les types Node, d'ou require()
// declare a la main.
declare function require(name: string): { readFileSync(fd: number, encoding: string): string }

function dechiffre(message: string, k: number): string {
  let resultat = ''
  for (let i = 0; i < message.length; i++) {
    const c: string = message[i]
    if (c >= 'A' && c <= 'Z') {
      const position: number = (c.charCodeAt(0) - 65 - k + 26) % 26
      resultat += String.fromCharCode(65 + position)
    } else {
      resultat += c
    }
  }
  return resultat
}

const lines: string[] = require('fs').readFileSync(0, 'utf8').split('\n')
const indice: string = lines[0].trim()
const message: string = lines[1].replace(/\s+$/, '')

for (let k = 0; k < 26; k++) {
  const clair: string = dechiffre(message, k)
  if (clair.indexOf(indice) !== -1) {
    console.log(clair)
    break
  }
}
