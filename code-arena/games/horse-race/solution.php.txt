<?php
// Course de Chevaux - solution de reference (PHP)
//
// Modele "batch" de Code Arena : on lit tout le cas de test sur stdin,
// on simule la course seconde par seconde (meme logique que referee.py),
// puis on ecrit les noms dans l'ordre d'arrivee, un par ligne.

$lines = [];
while (($line = fgets(STDIN)) !== false) {
    $line = trim($line);
    if ($line !== '') {
        $lines[] = $line;
    }
}

[$distance, $n] = array_map('intval', preg_split('/\s+/', $lines[0]));

$horses = [];
for ($i = 1; $i <= $n; $i++) {
    [$name, $speed, $stamina] = preg_split('/\s+/', $lines[$i]);
    $horses[] = [
        'name'    => $name,
        'speed'   => (int) $speed,
        'stamina' => (int) $stamina,
        'position'=> 0,
    ];
}

$finishOrder = [];
$finished = array_fill(0, $n, false);
$t = 0;
$maxSteps = $distance * 10; // garde-fou

while (count($finishOrder) < $n && $t < $maxSteps) {
    $t++;
    $newlyDone = [];

    for ($i = 0; $i < $n; $i++) {
        if ($finished[$i]) {
            continue;
        }
        if ($horses[$i]['stamina'] > 0) {
            $horses[$i]['position'] += $horses[$i]['speed'];
            $horses[$i]['stamina']--;
        } else {
            $horses[$i]['position'] += intdiv($horses[$i]['speed'], 2);
        }

        if ($horses[$i]['position'] >= $distance) {
            $finished[$i] = true;
            $newlyDone[] = $i;
        }
    }

    // Egalite : vitesse decroissante puis nom alphabetique
    usort($newlyDone, function ($a, $b) use ($horses) {
        if ($horses[$a]['speed'] !== $horses[$b]['speed']) {
            return $horses[$b]['speed'] <=> $horses[$a]['speed'];
        }
        return strcmp($horses[$a]['name'], $horses[$b]['name']);
    });

    foreach ($newlyDone as $i) {
        $finishOrder[] = $horses[$i]['name'];
    }
}

echo implode("\n", $finishOrder) . "\n";
