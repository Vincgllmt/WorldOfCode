// Course de Chevaux - solution de reference (JavaScript)
//
// Modele "batch" de Code Arena : on lit tout le cas de test sur stdin,
// on simule la course seconde par seconde (meme logique que referee.py),
// puis on ecrit les noms dans l'ordre d'arrivee, un par ligne.
const lines = require('fs')
  .readFileSync(0, 'utf8')
  .split('\n')
  .map((l) => l.trim())
  .filter((l) => l !== '')

const [distance, n] = lines[0].split(/\s+/).map(Number)

const horses = []
for (let i = 1; i <= n; i++) {
  const [name, speed, stamina] = lines[i].split(/\s+/)
  horses.push({ name, speed: Number(speed), stamina: Number(stamina), position: 0 })
}

const finishOrder = []
const finished = new Array(n).fill(false)
let t = 0
const maxSteps = distance * 10 // garde-fou

while (finishOrder.length < n && t < maxSteps) {
  t++
  const newlyDone = []

  for (let i = 0; i < n; i++) {
    if (finished[i]) continue
    if (horses[i].stamina > 0) {
      horses[i].position += horses[i].speed
      horses[i].stamina--
    } else {
      horses[i].position += Math.floor(horses[i].speed / 2)
    }

    if (horses[i].position >= distance) {
      finished[i] = true
      newlyDone.push(i)
    }
  }

  // Egalite : vitesse decroissante puis nom alphabetique
  newlyDone.sort((a, b) => {
    if (horses[a].speed !== horses[b].speed) return horses[b].speed - horses[a].speed
    return horses[a].name < horses[b].name ? -1 : horses[a].name > horses[b].name ? 1 : 0
  })

  for (const i of newlyDone) {
    finishOrder.push(horses[i].name)
  }
}

console.log(finishOrder.join('\n'))
