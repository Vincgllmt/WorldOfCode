<?php
// Le Maitre du Sudoku - solution de reference (PHP)
//
// Backtracking (retour sur trace) : on cherche une case vide, on y essaie chaque
// chiffre autorise, et on resout recursivement le reste de la grille. Si la suite
// echoue, on efface le chiffre et on essaie le suivant.

function autorise(array $grille, int $ligne, int $colonne, int $chiffre): bool
{
    for ($i = 0; $i < 9; $i++) {
        if ($grille[$ligne][$i] === $chiffre || $grille[$i][$colonne] === $chiffre) {
            return false;
        }
    }
    // Coin en haut a gauche du carre 3x3 qui contient la case.
    $l0 = 3 * intdiv($ligne, 3);
    $c0 = 3 * intdiv($colonne, 3);
    for ($l = $l0; $l < $l0 + 3; $l++) {
        for ($c = $c0; $c < $c0 + 3; $c++) {
            if ($grille[$l][$c] === $chiffre) {
                return false;
            }
        }
    }
    return true;
}

// La grille est passee par reference : les chiffres poses restent apres le retour.
function resoudre(array &$grille): bool
{
    for ($ligne = 0; $ligne < 9; $ligne++) {
        for ($colonne = 0; $colonne < 9; $colonne++) {
            if ($grille[$ligne][$colonne] === 0) {
                for ($chiffre = 1; $chiffre <= 9; $chiffre++) {
                    if (autorise($grille, $ligne, $colonne, $chiffre)) {
                        $grille[$ligne][$colonne] = $chiffre;
                        if (resoudre($grille)) {
                            return true;
                        }
                        $grille[$ligne][$colonne] = 0;
                    }
                }
                // Aucun chiffre ne convient : une hypothese precedente etait fausse.
                return false;
            }
        }
    }
    // Plus aucune case vide : la grille est resolue.
    return true;
}

$grille = [];
for ($l = 0; $l < 9; $l++) {
    $ligne = trim(fgets(STDIN));
    $grille[] = array_map(static fn (string $c): int => $c === '.' ? 0 : (int) $c, str_split($ligne));
}

resoudre($grille);

foreach ($grille as $ligne) {
    echo implode('', $ligne), "\n";
}
