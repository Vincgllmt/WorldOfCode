// Le Maitre du Sudoku - solution de reference (JavaScript)
//
// Backtracking (retour sur trace) : on cherche une case vide, on y essaie chaque
// chiffre autorise, et on resout recursivement le reste de la grille. Si la suite
// echoue, on efface le chiffre et on essaie le suivant.

function autorise(grille, ligne, colonne, chiffre) {
  for (let i = 0; i < 9; i++) {
    if (grille[ligne][i] === chiffre || grille[i][colonne] === chiffre) return false
  }
  // Coin en haut a gauche du carre 3x3 qui contient la case.
  const l0 = 3 * Math.floor(ligne / 3)
  const c0 = 3 * Math.floor(colonne / 3)
  for (let l = l0; l < l0 + 3; l++) {
    for (let c = c0; c < c0 + 3; c++) {
      if (grille[l][c] === chiffre) return false
    }
  }
  return true
}

function resoudre(grille) {
  for (let ligne = 0; ligne < 9; ligne++) {
    for (let colonne = 0; colonne < 9; colonne++) {
      if (grille[ligne][colonne] === 0) {
        for (let chiffre = 1; chiffre <= 9; chiffre++) {
          if (autorise(grille, ligne, colonne, chiffre)) {
            grille[ligne][colonne] = chiffre
            if (resoudre(grille)) return true
            grille[ligne][colonne] = 0
          }
        }
        // Aucun chiffre ne convient : une hypothese precedente etait fausse.
        return false
      }
    }
  }
  // Plus aucune case vide : la grille est resolue.
  return true
}

const lines = require('fs')
  .readFileSync(0, 'utf8')
  .split('\n')
  .map((l) => l.trim())
  .filter((l) => l !== '')

const grille = lines.slice(0, 9).map((l) => l.split('').map((c) => (c === '.' ? 0 : Number(c))))

resoudre(grille)
