// Le Maitre du Sudoku - solution de reference (Java)
//
// Backtracking (retour sur trace) : on cherche une case vide, on y essaie chaque
// chiffre autorise, et on resout recursivement le reste de la grille. Si la suite
// echoue, on efface le chiffre et on essaie le suivant.
import java.util.Scanner;

public class Main {
    static boolean autorise(int[][] grille, int ligne, int colonne, int chiffre) {
        for (int i = 0; i < 9; i++) {
            if (grille[ligne][i] == chiffre || grille[i][colonne] == chiffre) {
                return false;
            }
        }
        // Coin en haut a gauche du carre 3x3 qui contient la case.
        int l0 = 3 * (ligne / 3);
        int c0 = 3 * (colonne / 3);
        for (int l = l0; l < l0 + 3; l++) {
            for (int c = c0; c < c0 + 3; c++) {
                if (grille[l][c] == chiffre) {
                    return false;
                }
            }
        }
        return true;
    }

    static boolean resoudre(int[][] grille) {
        for (int ligne = 0; ligne < 9; ligne++) {
            for (int colonne = 0; colonne < 9; colonne++) {
                if (grille[ligne][colonne] == 0) {
                    for (int chiffre = 1; chiffre <= 9; chiffre++) {
                        if (autorise(grille, ligne, colonne, chiffre)) {
                            grille[ligne][colonne] = chiffre;
                            if (resoudre(grille)) {
                                return true;
                            }
                            grille[ligne][colonne] = 0;
                        }
                    }
                    // Aucun chiffre ne convient : une hypothese precedente etait fausse.
                    return false;
                }
            }
        }
        // Plus aucune case vide : la grille est resolue.
        return true;
    }

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int[][] grille = new int[9][9];
        for (int l = 0; l < 9; l++) {
            String ligne = scanner.next();
            for (int c = 0; c < 9; c++) {
                char ch = ligne.charAt(c);
                grille[l][c] = ch == '.' ? 0 : ch - '0';
            }
        }

        resoudre(grille);

        StringBuilder sortie = new StringBuilder();
        for (int[] ligne : grille) {
            for (int chiffre : ligne) {
                sortie.append(chiffre);
            }
            sortie.append('\n');
        }
        System.out.print(sortie);
    }
}
