// Mars Lander - Episode 2 : solution de reference (TypeScript)
//
// Modele "batch" de Code Arena : pas de boucle interactive. On lit l'etat initial
// une seule fois, et on ecrit d'un coup TOUTE la sequence de commandes "rotate power".
//
// Astuce : la physique est deterministe et connue. On simule donc la descente
// nous-memes (exactement comme le referee), en pilotant en boucle fermee a chaque
// tour ; on enregistre les commandes demandees et on les imprime. Rejouees en
// open-loop par le referee, elles reproduisent la meme trajectoire -> atterrissage.
//
// Notes sur le runtime TypeScript de Piston : il compile avec la bibliotheque ES5
// (ni Math.trunc ni Map) et sans les types Node, d'ou require() declare a la main
// et une troncature ecrite a la main.
declare function require(name: string): { readFileSync(fd: number, encoding: string): string }

const G = 3.711

// Troncature vers zero (Math.trunc n'existe pas avec la bibliotheque ES5).
function trunc(value: number): number {
  return value < 0 ? Math.ceil(value) : Math.floor(value)
}

// Arrondi "au plus proche, moities vers le pair" : c'est celui de Python, dont
// cette solution reproduit la trajectoire au bit pres.
function roundHalfEven(value: number): number {
  const lower: number = Math.floor(value)
  const frac: number = value - lower
  if (frac > 0.5) return lower + 1
  if (frac < 0.5) return lower
  return lower % 2 === 0 ? lower : lower + 1
}

function clamp(value: number, low: number, high: number): number {
  return Math.max(low, Math.min(high, value))
}

function radians(degrees: number): number {
  return (degrees * Math.PI) / 180
}

const lines: string[] = require('fs')
  .readFileSync(0, 'utf8')
  .split('\n')
  .map((l: string) => l.trim())
  .filter((l: string) => l !== '')

// --- Entree : surface puis etat initial ---
const n: number = parseInt(lines[0].split(/\s+/)[0], 10)
const surface: Array<[number, number]> = []
for (let i = 0; i < n; i++) {
  const parts: string[] = lines[1 + i].split(/\s+/)
  surface.push([Number(parts[0]), Number(parts[1])])
}

const state: number[] = lines[1 + n].split(/\s+/).map(Number)
let X: number = state[0]
let Y: number = state[1]
let hs: number = state[2]
let vs: number = state[3]
let fuel: number = trunc(state[4])
let rotate: number = trunc(state[5])
let power: number = trunc(state[6])

// --- Zone d'atterrissage : segment plat le plus large ---
let best = -1
let x1 = 0
let x2 = 0
let ly = 0
for (let i = 0; i < n - 1; i++) {
  const a: [number, number] = surface[i]
  const b: [number, number] = surface[i + 1]
  if (a[1] === b[1] && Math.abs(b[0] - a[0]) > best) {
    best = Math.abs(b[0] - a[0])
    x1 = Math.min(a[0], b[0])
    x2 = Math.max(a[0], b[0])
    ly = a[1]
  }
}
const cx: number = (x1 + x2) / 2

const out: string[] = []
for (let step = 0; step < 1200; step++) {
  const alt: number = Y - ly
  const centered: boolean = x1 + 150 < X && X < x2 - 150

  // Vitesse horizontale visee : 0 au-dessus de la zone, sinon viser le centre.
  const targetHs: number = centered ? 0 : clamp((cx - X) / 20, -25, 25)

  // Angle : annuler la vitesse horizontale ; redresser (0) en approche finale.
  let angle: number
  if (alt > 250) {
    angle = trunc(clamp(-(targetHs - hs) * 2, -40, 40))
  } else {
    angle = Math.abs(hs) <= 18 ? 0 : trunc(clamp(-hs * 2, -15, 15))
  }

  // Vitesse verticale visee : maintien d'altitude tant qu'on n'est pas aligne,
  // puis descente reguliere, douce pres du sol.
  let targetVs: number
  if (alt > 200 && !(centered && Math.abs(hs) < 15)) {
    targetVs = -4
  } else {
    targetVs = -Math.min(36, 12 + alt * 0.04)
  }

  // Puissance pour atteindre la vitesse verticale visee compte tenu de l'angle.
  const c: number = Math.cos(radians(angle))
  const powerReq: number = trunc(clamp(roundHalfEven((targetVs - vs + G) / Math.max(c, 0.4)), 0, 4))

  out.push(angle + ' ' + powerReq)

  // On avance l'etat avec EXACTEMENT la physique du referee (clamps inclus).
  const reqR: number = clamp(angle, -90, 90)
  const reqP: number = clamp(powerReq, 0, 4)
  rotate = Math.max(rotate - 15, Math.min(rotate + 15, reqR))
  power = Math.max(power - 1, Math.min(power + 1, reqP))
  if (power > fuel) power = trunc(fuel)
  fuel -= power
  const rad: number = radians(rotate)
  const ax: number = -Math.sin(rad) * power
  const ay: number = Math.cos(rad) * power - G
  const nX: number = X + hs + 0.5 * ax
  const nY: number = Y + vs + 0.5 * ay
  hs += ax
  vs += ay

  // Stop des qu'on touche la zone plane (ou qu'on sort du cadre).
  if (nY <= ly && x1 <= nX && nX <= x2) break
  if (nX < 0 || nX >= 7000 || nY >= 3000) break
  X = nX
  Y = nY
}

console.log(out.join('\n'))
