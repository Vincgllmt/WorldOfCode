<?php
// Mars Lander - Episode 2 : solution de reference (PHP)
//
// Modele "batch" de Code Arena : pas de boucle interactive. On lit l'etat initial
// une seule fois, et on ecrit d'un coup TOUTE la sequence de commandes "rotate power".
//
// Astuce : la physique est deterministe et connue. On simule donc la descente
// nous-memes (exactement comme le referee), en pilotant en boucle fermee a chaque
// tour ; on enregistre les commandes demandees et on les imprime. Rejouees en
// open-loop par le referee, elles reproduisent la meme trajectoire -> atterrissage.

const G = 3.711;

// Arrondi "au plus proche, moities vers le pair" : c'est celui de Python, dont
// cette solution reproduit la trajectoire au bit pres (round() de PHP arrondit
// les moities a l'oppose de zero, ce qui divergerait).
function roundHalfEven(float $value): int
{
    $lower = (int) floor($value);
    $frac = $value - $lower;
    if ($frac > 0.5) {
        return $lower + 1;
    }
    if ($frac < 0.5) {
        return $lower;
    }
    return $lower % 2 === 0 ? $lower : $lower + 1;
}

function clampValue(float $value, float $low, float $high): float
{
    return max($low, min($high, $value));
}

$lines = [];
while (($line = fgets(STDIN)) !== false) {
    $line = trim($line);
    if ($line !== '') {
        $lines[] = $line;
    }
}

// --- Entree : surface puis etat initial ---
$n = (int) preg_split('/\s+/', $lines[0])[0];
$surface = [];
for ($i = 0; $i < $n; $i++) {
    $parts = preg_split('/\s+/', $lines[1 + $i]);
    $surface[] = [(float) $parts[0], (float) $parts[1]];
}

$state = array_map('floatval', preg_split('/\s+/', $lines[1 + $n]));
$X = $state[0];
$Y = $state[1];
$hs = $state[2];
$vs = $state[3];
$fuel = (int) $state[4];
$rotate = (int) $state[5];
$power = (int) $state[6];

// --- Zone d'atterrissage : segment plat le plus large ---
$best = -1.0;
$x1 = 0.0;
$x2 = 0.0;
$ly = 0.0;
for ($i = 0; $i < $n - 1; $i++) {
    [$a, $b] = [$surface[$i], $surface[$i + 1]];
    if ($a[1] === $b[1] && abs($b[0] - $a[0]) > $best) {
        $best = abs($b[0] - $a[0]);
        $x1 = min($a[0], $b[0]);
        $x2 = max($a[0], $b[0]);
        $ly = $a[1];
    }
}
$cx = ($x1 + $x2) / 2.0;

$out = [];
for ($step = 0; $step < 1200; $step++) {
    $alt = $Y - $ly;
    $centered = $x1 + 150 < $X && $X < $x2 - 150;

    // Vitesse horizontale visee : 0 au-dessus de la zone, sinon viser le centre.
    $targetHs = $centered ? 0.0 : clampValue(($cx - $X) / 20.0, -25.0, 25.0);

    // Angle : annuler la vitesse horizontale ; redresser (0) en approche finale.
    if ($alt > 250) {
        $angle = (int) clampValue(-($targetHs - $hs) * 2, -40.0, 40.0);
    } else {
        $angle = abs($hs) <= 18 ? 0 : (int) clampValue(-$hs * 2, -15.0, 15.0);
    }

    // Vitesse verticale visee : maintien d'altitude tant qu'on n'est pas aligne,
    // puis descente reguliere, douce pres du sol.
    if ($alt > 200 && !($centered && abs($hs) < 15)) {
        $targetVs = -4.0;
    } else {
        $targetVs = -min(36.0, 12.0 + $alt * 0.04);
    }

    // Puissance pour atteindre la vitesse verticale visee compte tenu de l'angle.
    $c = cos(deg2rad($angle));
    $powerReq = (int) clampValue((float) roundHalfEven(($targetVs - $vs + G) / max($c, 0.4)), 0.0, 4.0);

    $out[] = $angle . ' ' . $powerReq;

    // On avance l'etat avec EXACTEMENT la physique du referee (clamps inclus).
    $reqR = (int) clampValue((float) $angle, -90.0, 90.0);
    $reqP = (int) clampValue((float) $powerReq, 0.0, 4.0);
    $rotate = (int) max($rotate - 15, min($rotate + 15, $reqR));
    $power = (int) max($power - 1, min($power + 1, $reqP));
    if ($power > $fuel) {
        $power = $fuel;
    }
    $fuel -= $power;
    $rad = deg2rad($rotate);
    $ax = -sin($rad) * $power;
    $ay = cos($rad) * $power - G;
    $nX = $X + $hs + 0.5 * $ax;
    $nY = $Y + $vs + 0.5 * $ay;
    $hs += $ax;
    $vs += $ay;

    // Stop des qu'on touche la zone plane (ou qu'on sort du cadre).
    if ($nY <= $ly && $x1 <= $nX && $nX <= $x2) {
        break;
    }
    if ($nX < 0 || $nX >= 7000 || $nY >= 3000) {
        break;
    }
    $X = $nX;
    $Y = $nY;
}

echo implode("\n", $out), "\n";
