// L'Escalier de briques - solution de reference (Java)
//
// Une boucle parcourt les marches de 1 a N. Pour chaque marche, une seconde
// boucle assemble la rangee brique par brique, puis on l'affiche.
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        int n = scanner.nextInt();

        for (int i = 1; i <= n; i++) {
            StringBuilder rangee = new StringBuilder();
            for (int j = 0; j < i; j++) {
                rangee.append('#');
            }
            System.out.println(rangee);
        }
    }
}
