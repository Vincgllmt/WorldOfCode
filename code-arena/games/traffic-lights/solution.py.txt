# La Route des feux - solution de reference (Python)
#
# On lit le nombre de feux, puis pour chaque feu sa couleur. Une condition
# choisit l'ordre a donner : PASSE au vert, RALENTIS a l'orange, STOP au rouge.

n = int(input())

for i in range(n):
    couleur = input().strip()
    if couleur == "VERT":
        print("PASSE")
    elif couleur == "ORANGE":
        print("RALENTIS")
    else:
        print("STOP")
