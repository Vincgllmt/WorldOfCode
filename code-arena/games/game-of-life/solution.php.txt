<?php
// La Boite de Petri - solution de reference (PHP)
//
// Jeu de la vie : chaque generation est calculee dans une NOUVELLE grille, en ne
// lisant que l'ancienne (toutes les cases changent "en meme temps"). Les cases
// hors de la grille comptent comme mortes.

function voisines(array $grille, int $x, int $y, int $largeur, int $hauteur): int
{
    $total = 0;
    for ($dy = -1; $dy <= 1; $dy++) {
        for ($dx = -1; $dx <= 1; $dx++) {
            if ($dx === 0 && $dy === 0) {
                continue;
            }
            $nx = $x + $dx;
            $ny = $y + $dy;
            if ($nx >= 0 && $nx < $largeur && $ny >= 0 && $ny < $hauteur && $grille[$ny][$nx] === '#') {
                $total++;
            }
        }
    }
    return $total;
}

function generationSuivante(array $grille, int $largeur, int $hauteur): array
{
    $suivante = [];
    for ($y = 0; $y < $hauteur; $y++) {
        $ligne = '';
        for ($x = 0; $x < $largeur; $x++) {
            $n = voisines($grille, $x, $y, $largeur, $hauteur);
            $vivante = $grille[$y][$x] === '#';
            $ligne .= ($n === 3 || ($vivante && $n === 2)) ? '#' : '.';
        }
        $suivante[] = $ligne;
    }
    return $suivante;
}

[$largeur, $hauteur, $generations] = array_map('intval', preg_split('/\s+/', trim(fgets(STDIN))));
$grille = [];
for ($y = 0; $y < $hauteur; $y++) {
    $grille[] = trim(fgets(STDIN));
}

for ($g = 0; $g < $generations; $g++) {
    $grille = generationSuivante($grille, $largeur, $hauteur);
}

echo implode("\n", $grille), "\n";
