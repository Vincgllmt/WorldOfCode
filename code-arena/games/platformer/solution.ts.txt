// Cameleon Bros - solution de reference (TypeScript)
//
// Chaque portail envoie un type de probleme sur la 1re ligne et ses donnees sur la
// 2e. On aiguille sur le type et on ecrit la reponse attendue, telle quelle.
// Attention : IS_NEGATIVE attend "True"/"False" avec une majuscule, UNIQUE
// deduplique les jetons tels qu'ils sont ecrits, et SORT trie par valeur.
//
// Note : le runtime TypeScript de Piston n'embarque pas les types Node, on
// declare donc require() a la main plutot que d'importer le module "fs".
declare function require(name: string): { readFileSync(fd: number, encoding: string): string }

const lines: string[] = require('fs')
  .readFileSync(0, 'utf8')
  .split('\n')
  .filter((l: string) => l.trim() !== '')

const problemType: string = lines.length > 0 ? lines[0].trim().toUpperCase() : ''
const args: string = lines.length > 1 ? lines[1].trim() : ''
const tokens: string[] = args === '' ? [] : args.split(/\s+/)
const numbers: number[] = tokens.map(Number)

if (problemType === 'SUCCESSOR') {
  console.log(numbers[0] + 1)
} else if (problemType === 'IS_NEGATIVE') {
  console.log(numbers[0] < 0 ? 'True' : 'False')
} else if (problemType === 'ABS') {
  console.log(Math.abs(numbers[0]))
} else if (problemType === 'MIN') {
  console.log(Math.min(numbers[0], numbers[1]))
} else if (problemType === 'SUM') {
  console.log(numbers.reduce((total: number, n: number) => total + n, 0))
} else if (problemType === 'REVERSE') {
  console.log(tokens.slice().reverse().join(' '))
} else if (problemType === 'UNIQUE') {
  const seen: string[] = []
  for (let i = 0; i < tokens.length; i++) {
    if (seen.indexOf(tokens[i]) === -1) seen.push(tokens[i])
  }
  console.log(seen.join(' '))
} else if (problemType === 'SORT') {
  console.log(numbers.slice().sort((a: number, b: number) => a - b).join(' '))
}
